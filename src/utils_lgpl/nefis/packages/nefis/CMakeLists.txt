# Select the correct hidden-CHARACTER-length ABI for src/f2c.c based on
# which *Fortran* compiler will actually be calling into this library --
# not which C/C++ compiler compiles it. See the BFtnLen comment in
# src/f2c.c: GNU Fortran and Flang pass an 8-byte size_t hidden length,
# Intel ifort/ifx pass a 4-byte int, and the two must not be confused (a
# wrong choice silently corrupts CHARACTER arguments rather than failing
# to build). This replaces an earlier version that inferred the ABI from
# `__GNUC__` in f2c.c itself, which is a C-compiler macro and happened to
# give the right answer on macOS only because Apple Clang also defines it.
function(nefis_set_hidden_strlen_definition target)
    if (CMAKE_Fortran_COMPILER_ID MATCHES "^(GNU|Flang|LLVMFlang)$")
        target_compile_definitions(${target} PRIVATE "D3D_FC_HIDDEN_STRLEN_IS_SIZE_T")
    elseif (CMAKE_Fortran_COMPILER_ID MATCHES "^(Intel|IntelLLVM)$")
        target_compile_definitions(${target} PRIVATE "D3D_FC_HIDDEN_STRLEN_IS_INT32")
    else()
        message(WARNING
            "nefis: unrecognized CMAKE_Fortran_COMPILER_ID "
            "'${CMAKE_Fortran_COMPILER_ID}' for target ${target} -- cannot "
            "determine whether this Fortran compiler passes hidden "
            "CHARACTER lengths as a 4-byte int (Intel-style) or an 8-byte "
            "size_t (GNU/Flang-style). Defaulting to the 4-byte-int "
            "convention (this codebase's historical default). If NEFIS "
            "calls start corrupting CHARACTER arguments (NEFIS error 5006 "
            "'This element type is not supported' is the known symptom), "
            "add this compiler ID to nefis_set_hidden_strlen_definition() "
            "in src/utils_lgpl/nefis/packages/nefis/CMakeLists.txt.")
        target_compile_definitions(${target} PRIVATE "D3D_FC_HIDDEN_STRLEN_IS_INT32")
    endif()
endfunction()


# Gather source files
file(GLOB headers include/*.h)
file(GLOB source ${src_path}/*.c
                 ${src_path}/*.cpp)
list(REMOVE_ITEM source ${CMAKE_CURRENT_SOURCE_DIR}/${src_path}/nef-xdr.c)

# Define library
set(library_name nefis)
add_subdirectory(test)

add_library(${library_name} ${headers}
                            ${source}
                            ${ini_version_file})

# Set additional dependencies
target_include_directories(${library_name}  PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include" ${version_include_dir})

# Set preprocessor definitions
if (UNIX)
    target_compile_definitions(${library_name} PRIVATE "PTR8;linux;LINUX64;_IA64_;_LIB;_CRT_SECURE_NO_WARNINGS")
endif(UNIX)
if (WIN32)
    target_compile_definitions(${library_name} PRIVATE "W64;PTR8;WIN64;_WIN64;_IA64_;_WINDOWS;_LIB;_CRT_SECURE_NO_WARNINGS")
endif(WIN32)

# f2c.c's BFtnLen (the C type used for the hidden length Fortran appends to
# every CHARACTER actual argument) must match the *Fortran* compiler's
# calling convention, not whichever C/C++ compiler compiles this library --
# GNU Fortran and LLVM Flang pass an 8-byte size_t; Intel ifort/ifx pass a
# 4-byte int (BInt4). Derive it from CMAKE_Fortran_COMPILER_ID here (see the
# BFtnLen comment in src/f2c.c) instead of inferring it from C-compiler
# predefined macros, which broke for any C/Fortran compiler combination
# other than the one this port happened to be built with.
nefis_set_hidden_strlen_definition(${library_name})

set_source_files_properties(${CMAKE_CURRENT_SOURCE_DIR}/src/*.c
                            ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp
                            PROPERTIES COMPILE_OPTIONS "${file_preprocessor_flag}")

	source_group(${visual_studio_source_group_name} FILES ${source})
	source_group("${visual_studio_source_group_name}\\Header Files" FILES ${headers})
	source_group("Version Files" FILES ${ini_version_file})
	
	# Create the folder structure in vfproj
	source_group(TREE ${CMAKE_CURRENT_SOURCE_DIR} FILES ${headers} ${source})
	set_target_properties (${library_name} PROPERTIES FOLDER utils_lgpl/nefis)
endif ()

string(TOUPPER ${CONFIGURATION_TYPE} configuration_type)
if ( ${configuration_type} STREQUAL "DELFT3D4" OR 
     ${configuration_type} STREQUAL "D3D4-TESTBENCH" OR
     ${configuration_type} STREQUAL "D3D4-SUITE" OR
     ${configuration_type} STREQUAL "TOOLS" )    
    set(library_name nefis_dll)

    add_library(${library_name} SHARED ${headers} ${source} ${ini_version_file})

    # Set additional dependencies
    target_include_directories(${library_name}  PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include" ${version_include_dir})

    # Set preprocessor definitions
    if (UNIX)
        target_compile_definitions(${library_name} PRIVATE "PTR8;linux;LINUX64;_IA64_;_LIB;_CRT_SECURE_NO_WARNINGS")    
    endif(UNIX)
    if (WIN32)
        target_compile_definitions(${library_name} PRIVATE "NEFIS_DLL;W64;PTR8;WIN64;_WIN64;_IA64_;_WINDOWS;_LIB;_CRT_SECURE_NO_WARNINGS")
    endif(WIN32)

    nefis_set_hidden_strlen_definition(${library_name})

    source_group("Source_files" FILES ${source})
    source_group("Header Files" FILES ${headers})
    source_group("Version Files" FILES ${ini_version_file})
  
    source_group(TREE ${CMAKE_CURRENT_SOURCE_DIR} FILES ${headers} ${source} ${ini_version_file})
    set_target_properties(${library_name} PROPERTIES FOLDER utils_lgpl/nefis_dll)
    set_target_properties(${library_name} PROPERTIES OUTPUT_NAME nefis)
    if(UNIX)
        set_target_properties(${library_name} PROPERTIES PREFIX "")
    endif()
  
    install(TARGETS ${library_name} RUNTIME DESTINATION bin
                                    LIBRARY DESTINATION lib
    )
endif()
